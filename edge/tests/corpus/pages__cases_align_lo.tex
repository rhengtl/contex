\section*{1 Piecewise definitions}

\[
H(x) = \begin{cases} 0 & \text{if } x < 0, \\ 1 & \text{if } x \geq 0. \end{cases}
\]

\begin{equation}
(a + b)^2 = a^2 + 2ab + b^2
\end{equation}
\begin{equation}
(a - b)^2 = a^2 - 2ab + b^2
\end{equation}
